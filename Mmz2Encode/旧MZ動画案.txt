コマンド
01-07, xx : プレーン, 個数
00,08-09  : Reserve
0Ah       : 中断
0Bh       : 終了
C000-FFFF : GVRAM Position

例)
C000h, 07h, 40, 16x3x40
C280h, 07h, 40, 16x3x40
C500h, 07h, 2, 16x3x2,
00, 07h, 2, 16x3x2,

;16x8単色を転送する
;DE: POSITION(2の倍数)
;HL: PATTERN
;破壊 BC, HL
;	LD	DE, POSITION
;	LD	HL, PATTERN
PUT16x8:
	PUSH	DE
	LD	(PUT16x8_RESTORE_STACK + 1), SP
	LD	SP, HL
	EX	DE, HL
	LD	BC, 78
	REPT 7
	POP	DE
	LD	(HL), E
	INC	L	; INC	HL (POSITIONが2の倍数の時のみ)
	LD	(HL), D
	INC	HL
	ADD	HL, BC
	ENDM
	POP	DE
	LD	(HL), E
	INC	L	; INC	HL (POSITIONが2の倍数の時のみ)
	LD	(HL), D
PUT16x8_RESTORE_STACK:
	LD	SP, 0000
	POP	DE
	RET
